# C Compilation and Disassembly

gcc -O2 hello.c -o hello
./hello
xxd hello

gcc -da -O2 hello.c -o hello

objdump -d -C -Mintel --disassemble=main hello
objdump -d -C -Mintel -s -j .text -j .rodata ./hello

# Java Compilation, Disassembly, and Class Loading

javac Hello.java
java Hello

xxd Hello.class
javap -c Hello.class
javap -v -c Hello.class

java -verbose:class Hello
